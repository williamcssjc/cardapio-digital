**Função `loadDrinks` — com tratamento diferenciado de erros:**
```typescript
async function loadDrinks() {
  setLoadingDrinks(true)
  setDrinksError(null)
  const supabase = createClient()

  // A aplicação depende exclusivamente de category_role = 'drinks'
  // Nunca usar o nome textual da categoria
  // limit(1) evita quebra caso mais de uma categoria esteja marcada como drinks
  // A unicidade por restaurante será implementada com a arquitetura multi-tenant
  const { data: categoryData, error: categoryError } = await supabase
    .from('categories')
    .select('id')
    .eq('category_role', 'drinks')
    .limit(1)
    .maybeSingle()

  if (categoryError) {
    console.error('[recepcao] Erro ao buscar categoria de bebidas:', categoryError)
    setDrinksError('Não foi possível carregar as bebidas.')
    setLoadingDrinks(false)
    return
  }

  if (!categoryData) {
    // Categoria não configurada — não é erro de banco
    console.warn('[recepcao] Nenhuma categoria com category_role = drinks encontrada.')
    setLoadingDrinks(false)
    return
  }

  const { data, error: itemsError } = await supabase
    .from('menu_items')
    .select('*') // substituir por campos reais após inspeção
    .eq('category_id', categoryData.id)
    .eq([coluna de disponibilidade real confirmada na inspeção], true)
    .order('name')

  if (itemsError) {
    console.error('[recepcao] Erro ao buscar bebidas:', itemsError)
    setDrinksError('Não foi possível carregar as bebidas.')
    setLoadingDrinks(false)
    return
  }

  // data vazio = categoria existe mas sem itens disponíveis — não é erro
  setDrinks(data ?? [])
  setLoadingDrinks(false)
}
```

Substituir `[coluna de disponibilidade real confirmada na inspeção]` pelo nome real encontrado.

**Função `handleConfirmDrink` — com try/catch obrigatório:**
```typescript
async function handleConfirmDrink(drink: MenuItem, qty: number) {
  setOrderState('submitting')
  setErrorMsg('')

  const { customerSessionId, tableSessionId, customer, context } = useSession.getState()

  try {
    const res = await fetch('/api/orders', {
      method: 'POST',
      headers: { 'Content-Type': 'application/json' },
      body: JSON.stringify({
        name: customer.name,
        table_num: context.tableNum || null,
        table_session_id: tableSessionId,
        customer_session_id: customerSessionId,
        created_by: 'customer',
        items: [{ id: drink.id, name: drink.name, price: drink.price, qty }],
        total: drink.price * qty,
      }),
    })

    if (!res.ok) {
      const text = await res.text()
      console.error('[recepcao] Erro HTTP ao enviar pedido:', res.status, text)
      setOrderState('error')
      setErrorMsg('Não foi possível enviar o pedido. Tente novamente.')
      return
    }

    const data = await res.json()
    const orderId = Number(data.id)

    // Usar o formato exato de CustomerOrder confirmado na inspeção
    const newOrder = {
      id: orderId,
      status: 'pending' as const,
      items: [{ id: drink.id, name: drink.name, price: drink.price, qty }],
      total: drink.price * qty,
      itemCount: qty,
      createdAt: data.created_at,
      tableNum: context.tableNum || null,
    }

    useOrderTracker.getState().addOrder(newOrder)
    useAccount.getState().addOrder(newOrder) // usar assinatura real confirmada na inspeção
    useSession.getState().setStatus('tracking')

    setConfirmedDrinkName(`${qty}x ${drink.name}`)
    setSelectedDrink(null)
    setQuantity(1)
    setOrderState('confirmed')

  } catch (err) {
    // Erro de rede ou exceção inesperada
    console.error('[recepcao] Exceção ao enviar pedido:', err)
    setOrderState('error')
    setErrorMsg('Erro de conexão. Verifique sua internet e tente novamente.')
  }
}
```
